
  <!--
    활성 + 보관 합집합. 채널별로 테이블 이름만 다르고 컬럼과 술어는 같다.
    The live + archive union. Only the table names differ per channel; columns and predicates do not.
  -->
  <sql id="messageSource">
    <choose>
      <when test="c.channel.name() == 'ALIMTALK'">
        SELECT ID, MSGKEY, STATUS, RSLT, MSG_RSLT, CALLBACK, PHONE, SERIALNUM,
               REQDATE, SENTDATE, REPORTDATE, 'QUE' AS TABLE_TYPE
          FROM KKO_MSG
        UNION ALL
        SELECT ID, MSGKEY, STATUS, RSLT, MSG_RSLT, CALLBACK, PHONE, SERIALNUM,
               REQDATE, SENTDATE, REPORTDATE, 'LOG' AS TABLE_TYPE
          FROM KKO_MSG_LOG
      </when>
      <otherwise>
        SELECT ID, MSGKEY, STATUS, RSLT, MSG_RSLT, CALLBACK, PHONE, SERIALNUM,
               REQDATE, SENTDATE, REPORTDATE, 'QUE' AS TABLE_TYPE
          FROM KKF_MSG
        UNION ALL
        SELECT ID, MSGKEY, STATUS, RSLT, MSG_RSLT, CALLBACK, PHONE, SERIALNUM,
               REQDATE, SENTDATE, REPORTDATE, 'LOG' AS TABLE_TYPE
          FROM KKF_MSG_LOG
      </otherwise>
    </choose>
  </sql>

  <select id="findMessages" resultMap="talkMessageRow">
    SELECT /* TalkMessageMapper.findMessages */
           A.SERIALNUM,
           A.MSGKEY,
           A.ID,
           A.STATUS,
           A.RSLT,
           <!--
             FIX D-T20: 원값과 사전 설명을 따로 반환한다. 레거시는 SQL 에서 이어 붙였고, 사전에
             없는 코드는 서브쿼리 NULL 이 전체를 NULL 로 만들어 화면이 비었다.
             FIX D-T20: the raw code and the dictionary text are returned separately. The legacy
             concatenated in SQL, where an unmatched code nulled the whole expression and blanked the
             field.
           -->
           (SELECT E.ERR_MSG FROM KKB_ERRCD_INFO E
             WHERE E.CHNL_DSNC = 'T' AND E.RSLT_CD = A.RSLT
             LIMIT 1) AS RSLT_TEXT,
           A.MSG_RSLT,
           (SELECT E.ERR_MSG FROM KKB_ERRCD_INFO E
             WHERE E.CHNL_DSNC = 'MSG' AND E.RSLT_CD = A.MSG_RSLT
             LIMIT 1) AS MSG_RSLT_TEXT,
           <!--
             FIX D-T6: masking(decrypt(...)) 를 최외곽에 적용한다. 레거시는 decrypt() 만 적용해
             평문 번호를 브라우저로 보냈다.
             FIX D-T6: masking(decrypt(...)) at the outermost level. The legacy applied decrypt() alone
             and sent plaintext numbers to the browser.
           -->
           masking(decrypt(A.CALLBACK)) AS CALLBACK,
           masking(decrypt(A.PHONE))    AS PHONE,
           to_char(A.REQDATE, 'YYYYMMDD') AS REQDATE,
           to_char(A.REQDATE, 'HH24MISS') AS REQTIME,
           to_char(A.SENTDATE, 'HH24MISS') AS SENTTIME,
           to_char(A.REPORTDATE, 'HH24MISS') AS REPORTTIME,
           A.TABLE_TYPE
      FROM (
        <include refid="messageSource"/>
      ) A
    <include refid="messageWhere"/>
     <!--
       전순서. 메시지키는 한 거래 안에서 유일하므로 동시각에도 순서가 정의된다 — 목록 화면의
       D-T10 과 같은 이유이며, 오프셋 페이징에서는 표시 성질이 아니라 정확성 전제다.
       A total order: the message key is unique within a transaction, so ties are still ordered — the same
       reasoning as D-T10 on the list screen, and under offset paging a correctness precondition rather
       than a display property.
     -->
     ORDER BY A.REQDATE DESC, A.MSGKEY DESC
     LIMIT #{c.size} OFFSET #{c.offset}
  </select>
